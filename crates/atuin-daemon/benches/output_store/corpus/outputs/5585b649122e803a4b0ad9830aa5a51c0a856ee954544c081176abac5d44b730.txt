macro_rules! include_trimmed {
    ($path:expr) => {
        include_str!($path).trim_ascii_end()
    };
}

macro_rules! include_shell {
    ($path:expr) => {
        include_trimmed!(concat!("shell/", $path))
    };
}

pub struct Bash<'a> {
    pub include_guard: &'a str,
    pub main: &'a str,
    pub preexec: &'a str,
}
